getMacroTilt() — Full Logic
function getMacroTilt(macro, sector) { // 1. Regime base tilt const baseTilts = { expansion: +8, reflation: +3, stagflation: -8, contraction: -12, }; let tilt = baseTilts[macro.regime_id] ?? 0; // 2. Recession penalty if (macro.recession_prob >= 60) tilt -= 5; // 3. Credit stress penalty if (macro.credit_stress >= 70) tilt -= 3; // 4. Sector modifier const growthSectors = ["Technology", "Comm Services", "Consumer Disc"]; const defensiveSectors = ["Utilities", "Consumer Staples", "Healthcare"]; if (growthSectors.includes(sector)) tilt += (tilt > 0 ? +4 : -4); if (defensiveSectors.includes(sector)) tilt = Math.round(tilt * 0.5); // 5. Clamp and return tilt = Math.max(-20, Math.min(+20, Math.round(tilt))); return { tilt, label, color, reasons }; }
